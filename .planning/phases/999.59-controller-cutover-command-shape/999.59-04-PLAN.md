class Stage {
 public:
  const ColorRGBA& getBackgroundColor() const;   // pure reads stay public
  const Camera&    getCamera() const;
 private:
  void setBackgroundColor_(const ColorRGBA&);     // virtual stays virtual
  void setCamera_(const Camera&);                 // virtual stays virtual
  void setEyeDistance_(float);
  void setFocalDistance_(float);
  void setFogIntensity_(float);
  friend class StageMutation;                     // THE SINGLE friend
};
class StageMutation {
 public:
  explicit StageMutation(Stage& s);
  StageMutation& backgroundColor(const ColorRGBA&);
  StageMutation& camera(const Camera&);
  StageMutation& eyeDistance(float);
  StageMutation& focalDistance(float);
  StageMutation& fogIntensity(float);
 private:
  Stage& stage_;
};
